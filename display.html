// ফায়ারবেস থেকে রিয়েলটাইম ডাটা সিঙ্ক (যখন এডমিন Next বাটনে চাপ দেবেন) db.ref('queue/current_calling').on('value', (snapshot) => { const callingSerial = snapshot.val(); if (callingSerial && callingSerial > 0) { // ১. ডিসপ্লে স্ক্রিনে নম্বর আপডেট const numElem = document.getElementById('display-number'); if (numElem) { numElem.innerText = '#' + String(callingSerial).padStart(3, '0'); } // ২. ডিং-ডং টোন ও ইংরেজি ভয়েস playDingDongAndSpeak(callingSerial); } }); function playDingDongAndSpeak(serialNumber) { const AudioCtx = window.AudioContext || window.webkitAudioContext; const audioCtx = new AudioCtx(); // Ding Sound const osc1 = audioCtx.createOscillator(); const gain1 = audioCtx.createGain(); osc1.type = 'sine'; osc1.frequency.setValueAtTime(659.25, audioCtx.currentTime); gain1.gain.setValueAtTime(0.4, audioCtx.currentTime); gain1.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5); osc1.connect(gain1); gain1.connect(audioCtx.destination); osc1.start(audioCtx.currentTime); osc1.stop(audioCtx.currentTime + 0.5); // Dong Sound const osc2 = audioCtx.createOscillator(); const gain2 = audioCtx.createGain(); osc2.type = 'sine'; osc2.frequency.setValueAtTime(523.25, audioCtx.currentTime + 0.3); gain2.gain.setValueAtTime(0.4, audioCtx.currentTime + 0.3); gain2.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.9); osc2.connect(gain2); gain2.connect(audioCtx.destination); osc2.start(audioCtx.currentTime + 0.3); osc2.stop(audioCtx.currentTime + 0.9); // English Voice: "Next 1", "Next 2"... setTimeout(() => { if ('speechSynthesis' in window) { window.speechSynthesis.cancel(); const msg = new SpeechSynthesisUtterance(`Next ${serialNumber}`); msg.lang = 'en-US'; msg.rate = 0.85; window.speechSynthesis.speak(msg); } }, 850); }
